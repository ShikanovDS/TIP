Задача 2


#include <iostream>

using namespace std;

int main() {
    int n = 4;
    
    unsigned long long total_result = 1;  
    unsigned long long current_fact = 1;  

    for (int i = 1; i <= n; i++) {
        current_fact = current_fact * i;            
        total_result = total_result * current_fact; 
    }

    cout << "Произведение факториалов для n = " << n << " равно: " << total_result << endl;

    return 0;
}
